    \end{whitemath}}

\begin{frame}
    
    \begin{whitemath}
        
        \begin{prooftree}
            \AXC{\(x: A \in \G\)}
            \UIC{\(\G \vdash x:A\)}
        \end{prooftree}
    \end{whitemath}
    \begin{whitemath}
        \begin{center}
            \AXC{\(\G \vdash M_1 : A_1\)}
            \AXC{\(\G \vdash M_2 : A_2\)}
            \BIC{\(\G \vdash (M_1, M_2) : A_1 \times A_2\)}
            \DP\\\ \\
            \AXC{\(\G \vdash M: A_1 \times A_2\)}
            \UIC{\(\G \vdash fst(M): A_1\)}
            \DP
            \qquad
            \AXC{\(\G \vdash M: A_1 \times A_2\)}
            \UIC{\(\G \vdash snd(M): A_2\)}
            \DP
        \end{center}
    \end{whitemath}
    \begin{whitemath}
    \begin{center}
        \AXC{\(\G, x:A \vdash M:B\)}
        \UIC{\(\G \vdash (x \mapsto M): A \rightarrow B\)}
        \DP\\\ \\
        \AXC{\(\G M_1: A \rightarrow B\)}
        \AXC{\(\G \vdash M_2: A\)}
        \BIC{\(\G \vdash M_1 M_2 : B\)}
        \DP
    \end{center}
        
    \end{whitemath}
\end{frame}
